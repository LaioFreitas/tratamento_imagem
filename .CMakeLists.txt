cmake_minimum_required(VERSION 3.20)

project(Processador_de_imagem VERSION 1.0 LANGUAGES CXX)
set(CMSKE_EXPORT_COMPILE_COMMSNDS ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
set(BUILD_opencv_videoio OFF CACHE INTERNAL "") # Evita compilar o cap_ffmpeg.cpp
set(WITH_FFMPEG OFF CACHE INTERNAL "")
set(BUILD_opencv_dnn OFF CACHE INTERNAL "")
# set(WITH_IPP OFF CACHE INTERNAL "")     # Desativa Intel IPP (Causa o erro do mean_ipp.cpp)
# set(WITH_ITT OFF CACHE INTERNAL "")     # Desativa Intel ITT
# set(WITH_LAPACK OFF CACHE INTERNAL "")
set(BUILD_EXAMPLES OFF CACHE INTERNAL "")
set(BUILD_TESTS OFF CACHE INTERNAL "")
set(BUILD_PERF_TESTS OFF CACHE INTERNAL "")
set(BUILD_opencv_apps OFF CACHE INTERNAL "")
set(BUILD_opencv_python3 OFF CACHE INTERNAL "")
FetchContent_Declare(opencv
                      GIT_REPOSITORY https://github.com/opencv/opencv.git
                      GIT_TAG 4.14.0
                      GIT_SHALLOW 0N)
FetchContent_MakeAvailable(opencv)
add_executable(Tratamento_img teste.cpp)
# target_link_libraries(Tratamento_img PRIVATE  ${OpenCV_LIBS})
